告别冷冰冰报错，生活化白话诊断

DEVELOPER QUICKSTART

30 秒集成至现有 React 项目

一行导入即可获得开箱即用的业务卡片流与双模工作台。

import { WorkflowIDE } from '@yiyi_zhang/reactflow-ui';
import '@yiyi_zhang/reactflow-ui/style.css';

export function CloudOpsStudio() {
  return (
    <div style={{ height: '100vh', width: '100vw' }}>
      <WorkflowIDE
        theme="dark"
        themeColor="blue"
        defaultUiMode="business" // 默认开启业务卡片流模式
        onSave={(workflowDef) => {
          // 保存为标准 Conductor 工作流定义
          console.log('保存工作流:', workflowDef);
        }}
      />
    </div>
  );
}
